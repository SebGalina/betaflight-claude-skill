# Preset: 3" cinewhoop / ducted
# Target: 3" ducted quad, ~300g AUW, 1404-1606 motors, 4S battery
# Betaflight 4.5.x — paste into CLI tab, then `save`
# Tuned for SMOOTH cinematic flight, not freestyle

# Motor & ESC
set motor_pwm_protocol = DSHOT600
set dshot_bidir = ON
set motor_poles = 12
set idle_min_rpm = 30
set motor_output_limit = 100

# Filters — tighter than freestyle because ducts cause more vibration
set gyro_lowpass_dyn_min_hz = 200
set gyro_lowpass_dyn_max_hz = 400
set dterm_lowpass_dyn_min_hz = 75
set dterm_lowpass_dyn_max_hz = 150
set dyn_notch_count = 4
set dyn_notch_q = 250
set dyn_notch_min_hz = 100
set dyn_notch_max_hz = 500

# Failsafe — DROP is fine for ducted (props protected)
set failsafe_procedure = DROP
set failsafe_delay = 4
set failsafe_off_delay = 10

# Profile 0
profile 0

# PID — slightly lower P, more I for smooth holds
set p_pitch = 40
set i_pitch = 100
set d_pitch = 35
set f_pitch = 140
set p_roll = 38
set i_roll = 95
set d_roll = 33
set f_roll = 135
set p_yaw = 45
set i_yaw = 110
set d_yaw = 0
set f_yaw = 130

# Feedforward — smoother for cinematic
set feedforward_smooth_factor = 40
set feedforward_jitter_factor = 10
set feedforward_boost = 10

# Rateprofile 0 — gentle cinematic rates
rateprofile 0
set rates_type = ACTUAL
set roll_rc_rate = 5
set roll_expo = 50
set roll_srate = 50
set pitch_rc_rate = 5
set pitch_expo = 50
set pitch_srate = 50
set yaw_rc_rate = 5
set yaw_expo = 40
set yaw_srate = 40

# RC smoothing — heavier for cine
set rc_smoothing = ON
set rc_smoothing_auto_factor = 45

save
