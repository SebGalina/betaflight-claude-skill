# Preset: 7" longrange
# Target: 7" quad, ~800g AUW, 2806-2807 motors 1300-1500 KV, 6S battery
# Betaflight 4.5.x — paste into CLI tab, then `save`
# Tuned for efficient cruise, NOT freestyle

# Motor & ESC
set motor_pwm_protocol = DSHOT300
set dshot_bidir = ON
set motor_poles = 14
set idle_min_rpm = 20
set motor_output_limit = 100

# Filters — slightly tighter for big slow props (lower noise freq)
set gyro_lowpass_dyn_min_hz = 180
set gyro_lowpass_dyn_max_hz = 400
set dterm_lowpass_dyn_min_hz = 70
set dterm_lowpass_dyn_max_hz = 150
set dyn_notch_count = 3
set dyn_notch_q = 300
set dyn_notch_min_hz = 80
set dyn_notch_max_hz = 500

# Failsafe — GPS_RESCUE only if GPS configured, otherwise LAND
set failsafe_procedure = GPS_RESCUE
set failsafe_delay = 4
set failsafe_off_delay = 10
# GPS rescue parameters (assume GPS connected on UART)
set gps_rescue_min_dth = 100
set gps_rescue_altitude_mode = MAX_ALT
set gps_rescue_ascend_rate = 500
set gps_rescue_descend_rate = 150
set gps_rescue_ground_speed = 500

# Profile 0
profile 0

# PID — lower P (heavy craft, less responsive), more I for stability
set p_pitch = 35
set i_pitch = 110
set d_pitch = 30
set f_pitch = 110
set p_roll = 33
set i_roll = 105
set d_roll = 28
set f_roll = 105
set p_yaw = 40
set i_yaw = 120
set d_yaw = 0
set f_yaw = 110

# Rateprofile 0 — moderate rates, predictable
rateprofile 0
set rates_type = ACTUAL
set roll_rc_rate = 8
set roll_expo = 30
set roll_srate = 60
set pitch_rc_rate = 8
set pitch_expo = 30
set pitch_srate = 60
set yaw_rc_rate = 7
set yaw_expo = 30
set yaw_srate = 50

save
