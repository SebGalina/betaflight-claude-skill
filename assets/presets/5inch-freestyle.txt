# Preset: 5" freestyle (default starting point)
# Target: 5" quad, ~600g AUW, 2207-class motors 1750-2400 KV, 6S battery
# Betaflight 4.5.x — paste into CLI tab, then `save`

# Motor & ESC
set motor_pwm_protocol = DSHOT600
set dshot_bidir = ON
set motor_poles = 14
set idle_min_rpm = 25
set motor_output_limit = 100

# Filters (open-ish — assumes clean build with bidir DSHOT)
set gyro_lowpass_dyn_min_hz = 250
set gyro_lowpass_dyn_max_hz = 500
set dterm_lowpass_dyn_min_hz = 100
set dterm_lowpass_dyn_max_hz = 200
set dyn_notch_count = 3
set dyn_notch_q = 300
set dyn_notch_min_hz = 100
set dyn_notch_max_hz = 600

# Failsafe — always DROP for freestyle
set failsafe_procedure = DROP
set failsafe_delay = 4
set failsafe_off_delay = 10

# Profile 0 — main freestyle profile
profile 0

# PID — slightly above stock for crisper feel
set p_pitch = 50
set i_pitch = 90
set d_pitch = 40
set f_pitch = 130
set p_roll = 47
set i_roll = 85
set d_roll = 37
set f_roll = 125
set p_yaw = 50
set i_yaw = 100
set d_yaw = 0
set f_yaw = 130

# Feedforward smoothing
set feedforward_transition = 0
set feedforward_averaging = AVG_2
set feedforward_smooth_factor = 25
set feedforward_jitter_factor = 7
set feedforward_boost = 15

# Rateprofile 0 — moderate freestyle rates
rateprofile 0
set rates_type = ACTUAL
set roll_rc_rate = 12
set roll_expo = 40
set roll_srate = 80
set pitch_rc_rate = 12
set pitch_expo = 40
set pitch_srate = 80
set yaw_rc_rate = 10
set yaw_expo = 40
set yaw_srate = 70

save
